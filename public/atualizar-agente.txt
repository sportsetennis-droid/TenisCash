# =====================================================================
# Atualiza o TenisCash Fiscal Agent DESTA loja a partir do servidor.
# Uso (PowerShell como ADMINISTRADOR, qualquer funcionario):
#   irm https://teniscash.com.br/atualizar-agente.txt | iex
# Le o token do proprio agente (C:\TenisCashAgent\.env), baixa os arquivos
# novos do central com verificacao de hash e reinicia a tarefa do agente.
# (ASCII puro de proposito - PowerShell 5.1 sem BOM quebra com acentos.)
# =====================================================================
$ErrorActionPreference = 'Stop'
$dir = if ($env:AGENT_DIR) { $env:AGENT_DIR } else { 'C:\TenisCashAgent' }
$base = 'https://teniscash.com.br/api/auth/agent-update'

if (-not (Test-Path (Join-Path $dir '.env'))) { Write-Host "ERRO: $dir\.env nao existe - este PC nao tem o agente instalado."; return }
$tok = if ($env:AGENT_TOKEN_OVERRIDE) { $env:AGENT_TOKEN_OVERRIDE } else {
  ((Get-Content (Join-Path $dir '.env')) | Where-Object { $_ -match '^AGENT_TOKEN=' } | Select-Object -First 1) -replace '^AGENT_TOKEN=', ''
}
if (-not $tok) { Write-Host 'ERRO: AGENT_TOKEN nao encontrado no .env do agente.'; return }
$h = @{ 'X-Agent-Token' = $tok.Trim() }

Write-Host 'Consultando versao no servidor...'
$man = Invoke-RestMethod -Uri "$base/manifest" -Headers $h -TimeoutSec 30
Write-Host ("Versao no servidor: " + $man.version)

foreach ($f in $man.files) {
  if ($f.name -eq 'package.json') { continue } # dependencia nova = reinstalacao manual, nao update
  $dest = Join-Path $dir $f.name
  if (Test-Path $dest) { Copy-Item $dest "$dest.bak" -Force }
  Invoke-WebRequest -Uri "$base/file/$($f.name)" -Headers $h -OutFile $dest -TimeoutSec 60 -UseBasicParsing
  $sha = (Get-FileHash $dest -Algorithm SHA256).Hash.ToLower()
  if ($sha -ne $f.sha256) {
    Write-Host ("ERRO: hash diferente em " + $f.name + " - restaurando backup e abortando.")
    if (Test-Path "$dest.bak") { Copy-Item "$dest.bak" $dest -Force }
    return
  }
  Write-Host ("  ok " + $f.name)
}

Write-Host 'Garantindo Tailscale + tunel publico (Funnel)...'
try { & tailscale up --timeout=30s 2>$null | Out-Null } catch {}
try { & tailscale funnel --bg 8765 2>$null | Out-Null; Write-Host '  tunel ok' } catch { Write-Host '  AVISO: funnel nao subiu - rode na mao: tailscale funnel --bg 8765' }

Write-Host 'Reiniciando o agente...'
try { Stop-ScheduledTask -TaskName TenisCashFiscalAgent -ErrorAction Stop } catch {}
Start-Sleep 2
$taskOk = $true
try { Start-ScheduledTask -TaskName TenisCashFiscalAgent -ErrorAction Stop } catch { $taskOk = $false; Write-Host 'AVISO: tarefa TenisCashFiscalAgent nao encontrada - inicie o agente manualmente (node index.js na pasta do agente).' }
if ($taskOk) {
  Start-Sleep 6
  try {
    $hl = Invoke-RestMethod -Uri 'http://localhost:8765/health' -TimeoutSec 10
    Write-Host ("PRONTO! Agente no ar - versao " + $hl.version)
  } catch { Write-Host 'Agente reiniciando... confira em alguns segundos: http://localhost:8765/health' }
}

# ---------------------------------------------------------------------
# SUPERVISOR (vigia que religa o agente + canal remoto, SEM AnyDesk).
# supervisor.js ja foi baixado acima (entrou no manifesto). Aqui so
# garantimos a tarefa TenisCashSupervisor e a iniciamos.
# ---------------------------------------------------------------------
if (Test-Path (Join-Path $dir 'supervisor.js')) {
  $supExists = $false
  try { Get-ScheduledTask -TaskName 'TenisCashSupervisor' -ErrorAction Stop | Out-Null; $supExists = $true } catch {}
  if (-not $supExists) {
    Write-Host 'Registrando o Supervisor (TenisCashSupervisor)...'
    $node = (Get-Command node -ErrorAction SilentlyContinue).Source
    if (-not $node) { $node = 'C:\Program Files\nodejs\node.exe' }
    try {
      $act = New-ScheduledTaskAction -Execute $node -Argument 'supervisor.js' -WorkingDirectory $dir
      $trg = New-ScheduledTaskTrigger -AtStartup
      $set = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries -RestartCount 999 -RestartInterval (New-TimeSpan -Minutes 1) -ExecutionTimeLimit ([TimeSpan]::Zero)
      Register-ScheduledTask -TaskName 'TenisCashSupervisor' -Action $act -Trigger $trg -Settings $set -RunLevel Highest -User 'SYSTEM' -Force | Out-Null
      Write-Host '  Supervisor registrado (SYSTEM, no boot, auto-restart).'
    } catch { Write-Host ('  AVISO: nao registrou o Supervisor: ' + $_.Exception.Message) }
  }
  try { Stop-ScheduledTask -TaskName 'TenisCashSupervisor' -ErrorAction SilentlyContinue } catch {}
  Start-Sleep 1
  try { Start-ScheduledTask -TaskName 'TenisCashSupervisor' -ErrorAction Stop; Write-Host 'Supervisor no ar (vigia o agente + obedece comandos do central).' } catch { Write-Host 'AVISO: Supervisor nao iniciou - confira a tarefa TenisCashSupervisor.' }
}

# ---------------------------------------------------------------------
# MONITOR (gravacao de seguranca do caixa, buffer rolante 36h).
# Roda na SESSAO DO USUARIO (tarefa no logon) - senao a captura de
# tela sai preta (isolamento de sessao do Windows). Ambiente AVISADO.
# ---------------------------------------------------------------------
if (Test-Path (Join-Path $dir 'monitor.js')) {
  # Compila o screencap.exe (captura via EXE — o AMSI bloqueia o MESMO codigo como script).
  if (Test-Path (Join-Path $dir 'screencap.cs')) {
    $csc = Get-ChildItem 'C:\Windows\Microsoft.NET\Framework64' -Filter csc.exe -Recurse -ErrorAction SilentlyContinue | Sort-Object FullName -Descending | Select-Object -First 1
    if (-not $csc) { $csc = Get-ChildItem 'C:\Windows\Microsoft.NET\Framework' -Filter csc.exe -Recurse -ErrorAction SilentlyContinue | Sort-Object FullName -Descending | Select-Object -First 1 }
    if ($csc) {
      $exe = Join-Path $dir 'screencap.exe'; $src = Join-Path $dir 'screencap.cs'
      try { & $csc.FullName /nologo /target:winexe "/out:$exe" /reference:System.Drawing.dll,System.Windows.Forms.dll "$src" | Out-Null } catch {}
      if (Test-Path $exe) { Write-Host '  screencap.exe compilado.' } else { Write-Host '  AVISO: screencap.exe nao compilou (csc falhou).' }
    } else { Write-Host '  AVISO: csc (.NET Framework) nao encontrado - screencap nao compilado.' }
  }
  $node2 = (Get-Command node -ErrorAction SilentlyContinue).Source
  if (-not $node2) { $node2 = 'C:\Program Files\nodejs\node.exe' }
  $usr = (Get-CimInstance Win32_ComputerSystem -ErrorAction SilentlyContinue).UserName
  $monExists = $false
  try { Get-ScheduledTask -TaskName 'TenisCashMonitor' -ErrorAction Stop | Out-Null; $monExists = $true } catch {}
  if ((-not $monExists) -and $usr) {
    Write-Host ('Registrando o Monitor na sessao de ' + $usr + '...')
    try {
      $act2 = New-ScheduledTaskAction -Execute $node2 -Argument 'monitor.js' -WorkingDirectory $dir
      $trg2 = New-ScheduledTaskTrigger -AtLogOn -User $usr
      $prin2 = New-ScheduledTaskPrincipal -UserId $usr -LogonType Interactive -RunLevel Limited
      $set2 = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries -RestartCount 999 -RestartInterval (New-TimeSpan -Minutes 1) -ExecutionTimeLimit ([TimeSpan]::Zero)
      Register-ScheduledTask -TaskName 'TenisCashMonitor' -Action $act2 -Trigger $trg2 -Principal $prin2 -Settings $set2 -Force | Out-Null
      Write-Host '  Monitor registrado (sessao do usuario, no logon).'
    } catch { Write-Host ('  AVISO: nao registrou o Monitor: ' + $_.Exception.Message) }
  }
  if (-not $usr) { Write-Host '  AVISO: ninguem logado agora - o Monitor inicia no proximo login do caixa.' }
  try { Stop-ScheduledTask -TaskName 'TenisCashMonitor' -ErrorAction SilentlyContinue } catch {}
  Start-Sleep 1
  try { Start-ScheduledTask -TaskName 'TenisCashMonitor' -ErrorAction Stop; Write-Host 'Monitor no ar (gravando a tela, buffer 36h, se renovando).' } catch { Write-Host 'Monitor inicia no proximo login do usuario.' }
}
